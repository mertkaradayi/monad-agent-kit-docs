Three lines to
your first swap

Import, configure, execute. The SDK handles gas, routing, and slippage.

typescript
import { MonadAgentKit } from "@monad-agent-kit/core"
import { defiPlugin } from "@monad-agent-kit/plugin-defi"

const agent = new MonadAgentKit({ privateKey })
agent.use(defiPlugin)

const result = await agent.execute("swap", {
  fromToken: "MON",
  toToken: "USDC",
  amount: "10",
})
